01: Online Test Graph Mastery ? Juspay Technologies

Complete algorithmic proofs, edge-case breakdowns, and optimal $O(N)$ C++20 and Python implementations for the 3 classic Juspay OA problems.

? The Functional Graph Invariant

Every problem in Juspay's Online Assessment operates on Directed Functional Graphs?graphs where every vertex has an out-degree of at most 1 (edges[i]). Every component consists of tree paths funneling into at most one directed cycle. Optimal time complexity is strictly $O(N)$, and space is $O(N)$.

1. Problem 1: Maximum Weight Node

1.1 Problem Statement

Given a directed graph of $N$ nodes ($0$ to $N-1$) where each node $i$ points to edges[i] (or -1 if no edge). The weight of node $u$ is the sum of all node indices $i$ that point directly to $u$:

$$\text{Weight}(u) = \sum_{i \in V, \text{edges}[i] = u} i$$

Find the node with the maximum weight. If multiple nodes have the same weight, return the one with the maximum index.

1.2 Complexity & Invariants

  • Time Complexity: $O(N)$ single linear pass.
  • Space Complexity: $O(N)$ accumulator array.
  • Edge Cases: Disconnected nodes (weight 0), self-loops (node contributes its own index to its weight), tie-breaking by highest node index.

1.3 Production C++20 Implementation

#include <iostream>
#include <vector>
#include <algorithm>

int maxWeightCell(int n, const std::vector<int>& edges) {
    std::vector<long long> weights(n, 0);

    for (int src = 0; src < n; ++src) {
        int dst = edges[src];
        if (dst >= 0 && dst < n) {
            weights[dst] += src;
        }
    }

    long long maxWeight = -1;
    int bestNode = -1;

    // Ties broken by choosing the larger index
    for (int i = 0; i < n; ++i) {
        if (weights[i] >= maxWeight) {
            maxWeight = weights[i];
            bestNode = i;
        }
    }

    return bestNode;
}

1.4 Production Python 3 Implementation

from typing import List

def max_weight_cell(n: int, edges: List[int]) -> int:
    weights = [0] * n
    for src in range(n):
        dst = edges[src]
        if 0 <= dst < n:
            weights[dst] += src

    max_weight = -1
    best_node = -1

    for node in range(n):
        if weights[node] >= max_weight:
            max_weight = weights[node]
            best_node = node

    return best_node

2. Problem 2: Nearest Meeting Cell (Shortest Meeting Point)

2.1 Problem Statement

Given two node indices $C_1$ and $C_2$, find a common node $u$ reachable from both such that the maximum of distances from $C_1$ to $u$ and $C_2$ to $u$ is minimized:

$$\min_{u \in \text{Reachable}(C_1) \cap \text{Reachable}(C_2)} \max(d(C_1, u), d(C_2, u))$$

If tied, return the node with the smallest index. If no common node is reachable, return -1.

2.2 Production C++20 Implementation

#include <iostream>
#include <vector>
#include <algorithm>
#include <climits>

std::vector<int> getDistances(int startNode, int n, const std::vector<int>& edges) {
    std::vector<int> dist(n, -1);
    std::vector<bool> visited(n, false);

    int curr = startNode;
    int d = 0;

    while (curr != -1 && !visited[curr]) {
        visited[curr] = true;
        dist[curr] = d++;
        curr = edges[curr];
    }
    return dist;
}

int nearestMeetingCell(int n, const std::vector<int>& edges, int c1, int c2) {
    std::vector<int> dist1 = getDistances(c1, n, edges);
    std::vector<int> dist2 = getDistances(c2, n, edges);

    int minMaxDist = INT_MAX;
    int bestNode = -1;

    for (int i = 0; i < n; ++i) {
        if (dist1[i] != -1 && dist2[i] != -1) {
            int currentMax = std::max(dist1[i], dist2[i]);
            if (currentMax < minMaxDist) {
                minMaxDist = currentMax;
                bestNode = i;
            }
        }
    }

    return bestNode;
}

2.3 Production Python 3 Implementation

from typing import List

def get_distances(start_node: int, n: int, edges: List[int]) -> List[int]:
    dist = [-1] * n
    visited = [False] * n
    curr = start_node
    d = 0
    while curr != -1 and not visited[curr]:
        visited[curr] = True
        dist[curr] = d
        d += 1
        curr = edges[curr]
    return dist

def nearest_meeting_cell(n: int, edges: List[int], c1: int, c2: int) -> int:
    dist1 = get_distances(c1, n, edges)
    dist2 = get_distances(c2, n, edges)

    min_max_dist = float('inf')
    best_node = -1

    for i in range(n):
        if dist1[i] != -1 and dist2[i] != -1:
            max_d = max(dist1[i], dist2[i])
            if max_d < min_max_dist:
                min_max_dist = max_d
                best_node = i

    return best_node

3. Problem 3: Largest Sum Cycle

3.1 Problem Statement

Given directed graph of $N$ nodes (each node has out-degree $\le 1$). Find the maximum sum of node indices in any directed cycle. If no cycle exists, return -1.

3.2 Algorithm: Kahn's Topological In-Degree Reduction

  1. Compute in-degrees for all $N$ nodes.
  2. Enqueue all nodes with in_degree == 0 into a BFS queue.
  3. Process queue, decrementing in-degree of downstream target nodes. Any target reaching 0 is enqueued. This peels away all non-cycle branches!
  4. Every node remaining with in_degree > 0 is guaranteed to belong to a directed cycle.
  5. Traverse each cycle, compute sum of indices, track maximum.

3.3 Production C++20 Implementation

#include <iostream>
#include <vector>
#include <queue>
#include <algorithm>

long long largestSumCycle(int n, const std::vector<int>& edges) {
    std::vector<int> inDegree(n, 0);

    for (int i = 0; i < n; ++i) {
        if (edges[i] != -1) {
            inDegree[edges[i]]++;
        }
    }

    std::queue<int> q;
    for (int i = 0; i < n; ++i) {
        if (inDegree[i] == 0) q.push(i);
    }

    std::vector<bool> visited(n, false);

    // Eliminate all branches feeding into cycles
    while (!q.empty()) {
        int u = q.front();
        q.pop();
        visited[u] = true;
        int v = edges[u];
        if (v != -1) {
            inDegree[v]--;
            if (inDegree[v] == 0) q.push(v);
        }
    }

    long long maxCycleSum = -1;

    // Remaining nodes strictly belong to cycles
    for (int i = 0; i < n; ++i) {
        if (!visited[i] && inDegree[i] > 0) {
            long long currentCycleSum = 0;
            int curr = i;

            while (!visited[curr]) {
                visited[curr] = true;
                currentCycleSum += curr;
                curr = edges[curr];
            }

            maxCycleSum = std::max(maxCycleSum, currentCycleSum);
        }
    }

    return maxCycleSum;
}

3.4 Production Python 3 Implementation

from typing import List
from collections import deque

def largest_sum_cycle(n: int, edges: List[int]) -> int:
    in_degree = [0] * n
    for dst in edges:
        if dst != -1:
            in_degree[dst] += 1

    q = deque([i for i in range(n) if in_degree[i] == 0])
    visited = [False] * n

    while q:
        u = q.popleft()
        visited[u] = True
        v = edges[u]
        if v != -1:
            in_degree[v] -= 1
            if in_degree[v] == 0:
                q.append(v)

    max_cycle_sum = -1

    for i in range(n):
        if not visited[i] and in_degree[i] > 0:
            cycle_sum = 0
            curr = i
            while not visited[curr]:
                visited[curr] = True
                cycle_sum += curr
                curr = edges[curr]
            max_cycle_sum = max(max_cycle_sum, cycle_sum)

    return max_cycle_sum
